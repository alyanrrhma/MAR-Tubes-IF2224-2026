program NestedAndComplex;
type
  Grade == (A, B, C, D, E);
  ScoreRange == 0..100;
var
  score, i, j: integer;
  grade: Grade;
  matrix: array[1..3] of integer;
procedure PrintMatrix(size: integer);
var
  k: integer;
begin
  for k := 1 to size do
    writeln(matrix[k])
end;
function GetGrade(s: integer): integer;
begin
  if s >= 80 then
    GetGrade := 1
  else if s >= 60 then
    GetGrade := 2
  else
    GetGrade := 3
end;
begin
  for i := 1 to 3 do begin
    matrix[i] := i * 10;
  PrintMatrix(3);
  score := 75;
  case GetGrade(score) of
    1: writeln('A');
    2: writeln('B');
    3: writeln('C')
  end;
  i := 1;
  while i <= 3 do begin
    j := 1;
    while j <= 3 do begin
      writeln(i * j);
      j := j + 1
    i := i + 1
  end
end.

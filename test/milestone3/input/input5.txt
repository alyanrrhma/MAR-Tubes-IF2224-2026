program LoopStatements;
var
  i, sum, factorial: integer;
begin
  sum := 0;
  i := 1;
  while i <= 10 do begin
    sum := sum + i;
    i := i + 1
  end;
  factorial := 1;
  i := 5;
  repeat
    factorial := factorial * i;
    i := i - 1
  until i == 0;
  for i := 10 downto 1 do
    writeln(i);
  writeln(sum);
  writeln(factorial)
end.
